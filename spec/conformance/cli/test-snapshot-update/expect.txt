# One update run rewrites the failing literal, fills the omitted one, and
# rewrites both snapshots of a test that has two, and exits 0. It leaves the
# passing literal and a call in an it_each case, which still fails.

exit: 1

run: hd test src/greet.hd
run: hd test --update src/rows.hd
exit: 1
run: hd test src/rows.hd
exit: 1
