# The native release artifact runs without translating its source module.
run: hd build --engine wasmtime --release
file: build/release/profile.cwasm
no-file: build/release/profile.wasm
run: hd --engine wasmtime build/release/profile.cwasm
stdout: module profile
run: hd run --engine wasmtime --release
stdout: module profile
# Explicit portable output preserves the recorded release profile.
run: hd build --engine wasmtime --release --format wasm
file: build/release/profile.wasm
run: hd --engine node build/release/profile.wasm
stdout: module profile
# Dev and Node outputs remain portable modules.
run: hd build --engine wasmtime
file: build/debug/profile.wasm
no-file: build/debug/profile.cwasm
run: hd build --engine node --release
file: build/release/profile.wasm
run: hd --engine wasmtime build/release/profile.wasm
stdout: module profile
# The selected source-file output uses the same native/portable choice.
run: hd build --engine wasmtime --release src/main.hd
file: build/release/files/main.cwasm
no-file: build/release/files/main.wasm
run: hd --engine wasmtime build/release/files/main.cwasm
stdout: module profile
run: hd build --engine wasmtime --release --format=wasm src/main.hd
file: build/release/files/main.wasm
run: hd --engine node build/release/profile.cwasm
