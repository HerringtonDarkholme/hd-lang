# A file under no root of a package is a single-file program that may use
# only std: no command loads the package's modules, so the printing task
# script `tasks/greet.hd` takes no part in them.
run: hd check tool.hd
run: hd tool.hd
stdout: hello, tool
run: hd check --format json uses_pkg.hd
exit: 101
stdout-json: [{"kind":"diagnostic","code":"unknown-module","file":"uses_pkg.hd","line":1},{"kind":"summary","errors":1,"status":101}]
run: hd tool.hd
stdout: hello, tool
# A task's script top level checks as `hd run` runs it (`cli.check.all`).
run: hd check --all
stderr: check result: ok.
run: hd check tasks/greet.hd
stderr: check result: ok.
run: hd run greet
stdout: hello from a task
# A use of `pkg` in a file under no root is an error in every command, and
# none of them names the task script.
run: hd uses_pkg.hd
exit: 101
stdout-json: []
stderr: error: uses_pkg.hd:1:1: unknown-module: no module named `shop`
run: hd test tested.hd
exit: 101
stderr: error: tested.hd:5:1: unknown-module: no module named `shop`
run: hd check --tests tested.hd
exit: 101
stderr: error: tested.hd:5:1: unknown-module: no module named `shop`
