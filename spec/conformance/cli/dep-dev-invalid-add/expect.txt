# A key in the other table is an error, not a move, and writes nothing.
# With the key from --name, hd add does not fetch either.
run: hd add --dev --name text github.com/acme/text@1.0.0
exit: 101
stderr: hd add: text is in [dependencies]; run `hd remove text` first
no-file: hd.sum
# Removing the last key also removes the dependency table header.
run: hd remove text
no-file: hd.sum
# hd add --dev still checks each target before it fetches, and writes nothing.
run: hd add --dev github.com/acme@1.0.0
exit: 101
no-file: hd.sum
run: hd add --dev --name 9fixtures github.com/acme/fixtures@1.0.0
exit: 101
no-file: hd.sum
