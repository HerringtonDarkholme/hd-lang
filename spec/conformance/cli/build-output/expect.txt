# `hd build FILE` writes the file's module only, under `files`.
run: hd build src/main.hd
file: build/debug/files/main.wasm
no-file: build/debug/main.wasm
no-file: build/debug/shop.wasm
# `--jobs N` builds the module as `hd build` does.
run: hd build --jobs 2
file: build/debug/shop.wasm
no-file: build/release
run: hd build --release
file: build/release/shop.wasm
run: hd build --release src/main.hd
file: build/release/files/main.wasm
# The removed `-o` form is a command-line error that writes nothing.
run: hd build src/main.hd -o x.wasm
exit: 101
stderr: error: `hd build` has no option `-o`
no-file: x.wasm
