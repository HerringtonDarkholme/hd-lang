# other's package is named money too, but the key money already requires
# the directory money. hd add never replaces it; its message suggests --name.
run: hd add --format json ./other
exit: 101
stdout-json: [{"kind":"diagnostic","code":"invalid-requirement","severity":"error","message":null},{"kind":"summary","errors":1,"status":101}]
# A key given by --name that requires another directory is the same error.
run: hd add --name money ./other
exit: 101
# Two targets that share one key are an error, and nothing is written.
run: hd add ./tax ./tax
exit: 101
run: hd remove tax
exit: 101
# A free key adds the second package.
run: hd add --name money2 ./other
stdout: money2 = { path = "other" }
run: hd run
stdout: 1100
